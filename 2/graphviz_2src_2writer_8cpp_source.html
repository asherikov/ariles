109 void writeNodeAndConnection(const Parameters::NodeOptions &node_options)
110 {
111 CPPUT_TRACE_FUNCTION;
112
113 const std::size_t stack_size = node_stack_.size();
114
115 CPPUT_ASSERT(0 < stack_size, "Internal error: stack must contain at least 2 entries.");
116
117 // node
118 back().actual_id_ = node_options.id_;
119
120 if (all_ids_.insert(back().actual_id_).second)
121 {
122 *output_stream_ << node_options.id_;
123 *output_stream_ << "[";
124 if (not node_options.label_.empty())
125 {
126 *output_stream_ << "label=\"" << node_options.label_ << "\"";
127 }
128 if (not node_options.options_.empty())
129 {
130 *output_stream_ << "," << node_options.options_;
131 }
132 *output_stream_ << "];\n";
133 }
134
135 // connection
136 if (stack_size > 1)
137 {
138 *output_stream_ //
139 << node_stack_[stack_size - 2].actual_id_ //
140 << "->" //
141 << back().actual_id_ << ";\n";
142 }
143 }

277 impl_->emplace(
