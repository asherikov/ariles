16 class ARILES2_VISIBILITY_ATTRIBUTE Any : public ariles2::DefaultBase
17 {
18#define ARILES2_ENTRIES(v) \

21#include ARILES2_INITIALIZE

23 protected:
24 bool isConsistent() const
25 {
26 if (("" != id_) && (nullptr != value_.get()))
27 {
28 return (true);
29 }
30
31 if (("" == id_) && (nullptr == value_.get()))
32 {
33 return (true);
34 }
35
36 return (false);

153 {
154 CPPUT_TRACE_FUNCTION;
155 CPPUT_ASSERT(
156 isConsistent(),

168 {

173 if ("" == id_)
174 {
175 CPPUT_ASSERT(parameters.allow_missing_entries_, "Id is empty, value cannot be read.");
176 }

179 build(id_);
