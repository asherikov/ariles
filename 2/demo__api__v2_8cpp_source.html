35namespace demo

37 class ArilesBaseClass
38 // must inherit from ariles2::DefaultBase
39 : public ariles2::DefaultBase
40 {
41// Declare entries, in this case two numbers
42#define ARILES2_ENTRIES(v) \
43 ARILES2_TYPED_ENTRY(v, real_member, double) \
44 ARILES2_TYPED_ENTRY_(v, integer_member, int)
45// underscore ^ indicates that the name of the entry must be
46// 'integer_member_' instead of 'integer_member', this is useful if your
47// naming convention requires trailing underscores for member variables.
48
49// Initialize ariles
50#include ARILES2_INITIALIZE
51
52 public:
53 virtual ~ArilesBaseClass() = default; // added to suppress compiler warnings
54
55 // This method is called every time you deserialize a class. If
56 // omitted, the default automatically generated method is used.
57 void arilesVisit(const ariles2::Defaults & /*visitor*/, const ariles2::Defaults::Parameters & /*param*/)
58 {
59 real_member = 0.0;
60 integer_member_ = 12;
61 }
62 };

73 class MyClass : public ArilesBaseClass, // no need to inherit from ariles2::DefaultBase directly.
74 public NonArilesBaseClass
75 {
76// Declare entries, in this case we indicate inheritance from another
77// Ariles class (ArilesBaseClass) and a member from a non-Ariles class
78// (NonArilesBaseClass)
79#define ARILES2_ENTRIES(v) \
80 ARILES2_PARENT(v, ArilesBaseClass) \
81 ARILES2_ENTRY_(v, eigen_vector)
82 // In this case ^ Ariles should not declare the inherited
83 // member, therefore we use 'ARILES2_ENTRY_' instead of 'ARILES2_TYPED_ENTRY_'.
84
85#include ARILES2_INITIALIZE
86
87
88 public:
89 ~MyClass() override = default; // added to suppress compiler warnings
90
91
92 void arilesVisit(const ariles2::Defaults &visitor, const ariles2::Defaults::Parameters &param)
93 {
94 // If you use your own method to initialize member variables,
95 // it is up to you to properly initialize all entries and
96 // parent classes.
97 // all parents at once
98 arilesVisitParents(visitor, param);
99 // or one by one (either option is sufficient)
100 ArilesBaseClass::arilesVisit(visitor, param);
101
102 // custom default values for some members
103 real_member = 100.0;
104 eigen_vector_.setZero();
105 }
106 };

138 // YAML
139 /*
140 * When you serialize `my_container_class` to YAML you get the following:
141 * -----
142 MyContainerClass:
143 my_class_vector:
144 - real_member: 100
145 integer_member: 12
146 eigen_vector: [0, 0, 0]
147 * -----
148 * Note that the trailing underscores are omitted for all members. This
149 * applies to all supported representations.
150 */
151 {
152 // You can read and write YAML configuration files as follows:
153 ariles2::apply<ariles2::yaml_cpp::Writer>("config.yaml", my_container_class);
154 ariles2::apply<ariles2::yaml_cpp::Reader>("config.yaml", my_container_class);
155
156 // Sometimes it may be useful to dump configuration to std::cout
157 ariles2::apply<ariles2::yaml_cpp::Writer>(std::cout, my_container_class);
158
159 // Adjust visitor parameters
160 ariles2::yaml_cpp::Writer::Parameters parameters;
161 parameters.get<ariles2::write::Parameters>().allow_missing_entries_ = true;
162 ariles2::apply<ariles2::yaml_cpp::Writer>(std::cout, my_container_class, parameters);
163
164 // In some situations it is more convenient to instantiate Reader and
165 // Writer classes explicitly, e.g., if you keep configurations of several
166 // classes in the same file
167 ariles2::yaml_cpp::Writer writer("config.yaml");
168 ariles2::apply(writer, my_container_class);
169
170 ariles2::yaml_cpp::Reader reader("config.yaml");
171 ariles2::apply(reader, my_container_class);
172 }
173
174
175 // ROS parameter server
176 {
177 ros::NodeHandle nh;
178
179 // read/write
180 ariles2::apply<ariles2::rosparam::Writer>(nh, my_container_class);
181 ariles2::apply<ariles2::rosparam::Reader>(nh, my_container_class);
182 // parameters can be uploaded to parameter server in advance using
183 // roslaunch, see http://wiki.ros.org/roslaunch/XML/rosparam
184
185 // read/write with namespace
186 ariles2::apply<ariles2::rosparam::Writer>(nh, my_container_class, "/some_namespace/");
187 ariles2::apply<ariles2::rosparam::Reader>(nh, my_container_class, "/some_namespace/");
188
189 // Reader / Writer classes
190 ariles2::rosparam::Writer writer(nh);
191 ariles2::apply(writer, my_container_class);
192
193 ariles2::rosparam::Reader reader(nh);
194 ariles2::apply(reader, my_container_class);
195 }
196
197
198 // Octave
199 {
200 // Octave visitor supports only writing
201 ariles2::apply<ariles2::octave::Writer>("debug.m", my_container_class);
202 // the generated file can later be loaded in Octave with
203 // 'source debug.m' for debugging
204 }
205 }
206 catch (const std::exception &e)
207 {
208 std::cout << e.what() << std::endl;
209 return (EXIT_FAILURE);
