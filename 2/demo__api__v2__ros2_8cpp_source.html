36 class ArilesBaseClass
37 // must inherit from ariles2::DefaultBase
38 : public ariles2::DefaultBase
39 {
40// Declare entries, in this case two numbers
41#define ARILES2_ENTRIES(v) \
42 ARILES2_TYPED_ENTRY(v, real_member, double) \
43 ARILES2_TYPED_ENTRY_(v, integer_member, int)
44// underscore ^ indicates that the name of the entry must be
45// 'integer_member_' instead of 'integer_member', this is useful if your
46// naming convention requires trailing underscores for member variables.
47
48// Initialize ariles
49#include ARILES2_INITIALIZE
50
51 public:
52 virtual ~ArilesBaseClass() = default; // added to suppress compiler warnings

54 // This method is called every time you deserialize a class. If
55 // omitted, the default automatically generated method is used.
56 void arilesVisit(const ariles2::Defaults & /*visitor*/, const ariles2::Defaults::Parameters & /*param*/)
57 {
58 real_member = 0.0;
59 integer_member_ = 12;
60 }

72 class MyClass : public ArilesBaseClass, // no need to inherit from ariles2::DefaultBase directly.
73 public NonArilesBaseClass
74 {
75// Declare entries, in this case we indicate inheritance from another
76// Ariles class (ArilesBaseClass) and a member from a non-Ariles class
77// (NonArilesBaseClass)
78#define ARILES2_ENTRIES(v) \
79 ARILES2_PARENT(v, ArilesBaseClass) \
80 ARILES2_ENTRY_(v, eigen_vector)
81 // Here ^ Ariles should not declare the inherited member, therefore we
82 // use 'ARILES2_ENTRY_' instead of 'ARILES2_TYPED_ENTRY_'.
83
84#include ARILES2_INITIALIZE
85
86
87 public:
88 ~MyClass() override = default; // added to suppress compiler warnings
89
90
91 void arilesVisit(const ariles2::Defaults &visitor, const ariles2::Defaults::Parameters &param)
92 {
93 // If you use your own method to initialize member variables,
94 // it is up to you to properly initialize all entries and
95 // parent classes.
96 // all parents at once
97 arilesVisitParents(visitor, param);
98 // or one by one (either option is sufficient)
99 ArilesBaseClass::arilesVisit(visitor, param);
100
101 // custom default values for some members
102 real_member = 100.0;
103 eigen_vector_.setZero();
104 }

129// run with "demo_api_v2_ros2 --ros-args --params-file /demo_api_v2_ros2.yaml"
130int main(int argc, char *argv[])
131{
132 try
133 {
134 rclcpp::init(argc, argv);
135 rclcpp::Node::SharedPtr nh = rclcpp::Node::make_shared(
136 "demo_api_v2_ros2",
137 // although ariles provides Declarator visitor, declaring
138 // parameters in general case is not possible since generic
139 // arrays are stored as maps with indices used as keys: size of
140 // such array is not know in advance so it is not possible to
141 // declare all the necessary indices
142 rclcpp::NodeOptions().allow_undeclared_parameters(true).automatically_declare_parameters_from_overrides(
143 true));
144
145 // read parameters loaded from file (demo_api_v2_ros2.yaml)
146 {
147 MyContainerClass my_container_class;
148
149 // access members as usual
150 my_container_class.my_class_vector_.emplace_back();
151 ariles2::apply<ariles2::Defaults>(my_container_class.my_class_vector_[0]);
152
153 /*
154 * output
155 my_class_vector:
156 - real_member: 100
157 integer_member: 12
158 eigen_vector: [0, 0, 0]
159 ptr:
160 is_null: true
161 */
162 ariles2::apply<ariles2::yaml_cpp::Writer>(std::cout, my_container_class);
163
164 // see demo_api_v2_ros2.yaml
165 ariles2::apply<ariles2::ros2param::Reader>(nh->get_node_parameters_interface(), my_container_class);
166
167 /*
168 * output
169 my_class_vector:
170 - real_member: 100
171 integer_member: 12
172 eigen_vector: [0, 0, 0]
173 - real_member: 110
174 integer_member: 1
175 eigen_vector: [0, 1, 0]
176 - real_member: 111
177 integer_member: 21
178 eigen_vector: [1, 1, 0]
179 ptr:
180 is_null: true
181 */
182 ariles2::apply<ariles2::yaml_cpp::Writer>(std::cout, my_container_class);
183 }
184
185 // writing & reading
186 {
187 MyContainerClass my_container_class;
188 my_container_class.ptr_ = std::make_shared<MyClass>();
189 ariles2::apply<ariles2::Defaults>(*my_container_class.ptr_);
190
191 my_container_class.my_class_vector_.emplace_back();
192 ariles2::apply<ariles2::Defaults>(my_container_class.my_class_vector_[0]);
193 my_container_class.my_class_vector_[0].real_member = 200;
194
195 ariles2::apply<ariles2::ros2param::Writer>(nh->get_node_parameters_interface(), my_container_class);
196 ariles2::apply<ariles2::ros2param::Reader>(nh->get_node_parameters_interface(), my_container_class);
197
198 /*
199 * output:
200 * - note that parameters loaded from file are preserved
201 * - it would be nice to drop old parameters, but it is impossible to undeclare them
202 my_class_vector:
203 - real_member: 200 # overridden
204 integer_member: 12
205 eigen_vector: [0, 0, 0]
206 - real_member: 110
207 integer_member: 1
208 eigen_vector: [0, 1, 0]
209 - real_member: 111
210 integer_member: 21
211 eigen_vector: [1, 1, 0]
212 ptr:
213 is_null: false # overridden
214 value: # added
215 real_member: 100
216 integer_member: 12
217 eigen_vector: [0, 0, 0]
218 */
219 ariles2::apply<ariles2::yaml_cpp::Writer>(std::cout, my_container_class);
220 }
221
222 // declaring
223 // - hardly useful, is going to declare my_class_vector_ to have only one member
224 // - declarator does not read parameters into an Ariles class, a
225 // separate pass with Reader visitor is needed

227 MyContainerClass my_container_class;
228 my_container_class.my_class_vector_.emplace_back();
229 ariles2::apply<ariles2::Defaults>(my_container_class.my_class_vector_[0]);

231 }

240 }

245 }
246 }
247 }

250 std::cout << e.what() << std::endl;
251 return (EXIT_FAILURE);

254 return (EXIT_SUCCESS);
