134  bool Reader::startMapEntry(const std::string &child_name)
135  {
136  CPPUT_TRACE_FUNCTION;
137  CPPUT_TRACE_VALUE(child_name);
138  if (impl_->empty())
139  {
140  for (std::size_t i = 0; i < impl_->handles_.size(); ++i)
141  {
142  if (::msgpack::type::MAP == impl_->handles_[i]->get().type)
143  {
144  if (child_name == impl_->handles_[i]->get().via.map.ptr[0].key.as<std::string>())
145  {
146  if (::msgpack::type::MAP == impl_->handles_[i]->get().via.map.ptr[0].val.type)
147  {
148  impl_->emplace(&(impl_->handles_[i]->get().via.map.ptr[0].val));
149  return (true);
150  }
151  }
152  }
153  }
154  }
155  else
156  {
157  if (::msgpack::type::MAP == impl_->getRawNode().type)
158  {
159  for (std::size_t i = 0; i < impl_->getRawNode().via.map.size; ++i)
160  {
161  if (child_name == impl_->getRawNode().via.map.ptr[i].key.as<std::string>())
162  {
163  impl_->emplace(&(impl_->getRawNode().via.map.ptr[i].val));
164  return (true);
165  }
166  }
167  }
168  }
169 
170  return (false);
